public class SistemaRegistroUI extends JFrame {

    // ── Servicios compartidos ──────────────────────────────────────────
    private static final ClienteServicio  clienteServicio  = new ClienteServicio();
    private static final ProductoServicio productoServicio = new ProductoServicio();
    private static final VentaServicio    ventaServicio    = new VentaServicio();
    private static int contadorVentas = 0;

    // CardLayout para navegar entre pantallas
    private final CardLayout cardLayout    = new CardLayout();
    private final JPanel     panelCentral = new JPanel(cardLayout);

    // ============================================================
    //  CONSTRUCTOR

    // ============================================================

    private JPanel panelMenu() {

        titulo.setFont(titulo.getFont().deriveFont(Font.BOLD, 18f));
        panel.add(titulo, BorderLayout.NORTH);

        JPanel botones = new JPanel(new GridLayout(2, 2, 10, 10));
        botones.setBorder(BorderFactory.createEmptyBorder(30, 60, 30, 60));

        JButton btnClientes    = new JButton("Gestion de Clientes");
        JButton btnProductos   = new JButton("Gestion de Productos");
        JButton btnNuevaVenta  = new JButton("Registrar Venta");
        JButton btnListaVentas = new JButton("Consultar Ventas");

        btnClientes   .addActionListener(e -> cardLayout.show(panelCentral, "clientes"));
        btnProductos  .addActionListener(e -> cardLayout.show(panelCentral, "productos"));
        btnNuevaVenta .addActionListener(e -> cardLayout.show(panelCentral, "ventas"));
        btnListaVentas.addActionListener(e -> cardLayout.show(panelCentral, "listaVentas"));

        botones.add(btnClientes);
        botones.add(btnProductos);
        botones.add(btnNuevaVenta);
        botones.add(btnListaVentas);

        panel.add(botones, BorderLayout.CENTER);
        return panel;

    // ============================================================
    //  P2 – GESTION DE CLIENTES
    // ============================================================
    private JPanel panelClientes() {
        JPanel panel = new JPanel(new BorderLayout(5, 5));
        panel.setBorder(BorderFactory.createEmptyBorder(10, 10, 10, 10));

        // Formulario
        JPanel form = new JPanel(new GridLayout(4, 2, 5, 5));
        form.setBorder(BorderFactory.createTitledBorder("Registrar Cliente"));

        JTextField txtCodigo = new JTextField();
        JTextField txtNombre = new JTextField();
        JTextField txtEmail  = new JTextField();
        JButton    btnReg    = new JButton("Registrar");
        JButton    btnBuscar = new JButton("Buscar por codigo");

        form.add(new JLabel("Codigo:"));  form.add(txtCodigo);
        form.add(new JLabel("Nombre:"));  form.add(txtNombre);
        form.add(new JLabel("Email:"));   form.add(txtEmail);
        form.add(btnReg);                 form.add(btnBuscar);

        // Tabla
        String[] cols = {"Codigo", "Nombre", "Email"};
        DefaultTableModel modelo = new DefaultTableModel(cols, 0) {
            @Override public boolean isCellEditable(int r, int c) { return false; }
        };
        JTable tabla = new JTable(modelo);
        JScrollPane scroll = new JScrollPane(tabla);
        scroll.setBorder(BorderFactory.createTitledBorder("Clientes registrados"));

        JPanel sur = new JPanel(new FlowLayout(FlowLayout.RIGHT));
        JButton btnActualizar = new JButton("Actualizar lista");
        sur.add(new JButton("Volver al menu") {{
            addActionListener(e -> cardLayout.show(panelCentral, "menu"));
        }});
        sur.add(btnActualizar);

        // Eventos
        Runnable refrescar = () -> {
            modelo.setRowCount(0);
            for (var c : clienteServicio.getClientes())
                modelo.addRow(new Object[]{c.getCodigo(), c.getNombre(),
                        c.getEmail() == null ? "" : c.getEmail()});
        };

        btnReg.addActionListener(e -> {
            String cod   = txtCodigo.getText().trim();
            String nom   = txtNombre.getText().trim();
            String email = txtEmail.getText().trim();
            if (cod.isEmpty() || nom.isEmpty()) {
                JOptionPane.showMessageDialog(panel, "Codigo y Nombre son obligatorios.");
                return;
            }
            try {
                int codigo = Integer.parseInt(cod);
                if (email.isEmpty()) clienteServicio.registrar(codigo, nom);
                else                 clienteServicio.registrar(codigo, nom, email);
                JOptionPane.showMessageDialog(panel, "Cliente registrado correctamente.");
                txtCodigo.setText(""); txtNombre.setText(""); txtEmail.setText("");
                refrescar.run();
            } catch (NumberFormatException ex) {
                JOptionPane.showMessageDialog(panel, "El codigo debe ser numerico.");
            } catch (ClienteException ex) {
                JOptionPane.showMessageDialog(panel, ex.getMessage());
            }
        });

        btnBuscar.addActionListener(e -> {
            String cod = txtCodigo.getText().trim();
            if (cod.isEmpty()) { JOptionPane.showMessageDialog(panel, "Ingrese el codigo."); return; }
            try {
                var c = clienteServicio.buscarPorCodigo(Integer.parseInt(cod));
                if (c == null) JOptionPane.showMessageDialog(panel, "Cliente no encontrado.");
                else {
                    txtNombre.setText(c.getNombre());
                    txtEmail.setText(c.getEmail() == null ? "" : c.getEmail());
                }
            } catch (NumberFormatException ex) {
                JOptionPane.showMessageDialog(panel, "El codigo debe ser numerico.");
            }
        });

        btnActualizar.addActionListener(e -> refrescar.run());

        panel.add(form,   BorderLayout.NORTH);
        panel.add(scroll, BorderLayout.CENTER);
        panel.add(sur,    BorderLayout.SOUTH);

        refrescar.run();
        return panel;
    }

    // ============================================================
    //  P3 – GESTION DE PRODUCTOS
    // ============================================================
    private JPanel panelProductos() {
        JPanel panel = new JPanel(new BorderLayout(5, 5));
        panel.setBorder(BorderFactory.createEmptyBorder(10, 10, 10, 10));

        // Formulario
        JPanel form = new JPanel(new GridLayout(5, 2, 5, 5));
        form.setBorder(BorderFactory.createTitledBorder("Registrar Producto"));

        JTextField txtCodigo  = new JTextField();
        JTextField txtNombre  = new JTextField();
        JTextField txtPrecio  = new JTextField();
        JComboBox<String> cbTipo = new JComboBox<>(new String[]{"Electronico","Alimento","Ropa"});
        JCheckBox chkExtra    = new JCheckBox("Tiene garantia / Es perecible / En temporada");
        JTextField txtTalla   = new JTextField();
        JButton btnReg        = new JButton("Registrar Producto");

        cbTipo.addActionListener(e -> {
            String t = (String) cbTipo.getSelectedItem();
            if ("Electronico".equals(t)) chkExtra.setText("Tiene garantia");
            else if ("Alimento".equals(t)) chkExtra.setText("Es perecible");
            else chkExtra.setText("Esta en temporada");
        });

        form.add(new JLabel("Codigo:"));  form.add(txtCodigo);
        form.add(new JLabel("Nombre:"));  form.add(txtNombre);
        form.add(new JLabel("Precio:"));  form.add(txtPrecio);
        form.add(new JLabel("Tipo:"));    form.add(cbTipo);
        form.add(chkExtra);               form.add(new JPanel() {{ // talla inline
            setLayout(new FlowLayout(FlowLayout.LEFT, 2, 0));
            add(new JLabel("Talla (Ropa):"));
            add(txtTalla);
        }});

        // Tabla
        String[] cols = {"Codigo","Nombre","Tipo","Precio Final"};
        DefaultTableModel modelo = new DefaultTableModel(cols, 0) {
            @Override public boolean isCellEditable(int r, int c) { return false; }
        };
        JTable tabla = new JTable(modelo);
        JScrollPane scroll = new JScrollPane(tabla);
        scroll.setBorder(BorderFactory.createTitledBorder("Productos registrados"));

        JPanel sur = new JPanel(new FlowLayout(FlowLayout.RIGHT));
        JButton btnActualizar = new JButton("Actualizar lista");
        sur.add(new JButton("Volver al menu") {{
            addActionListener(e -> cardLayout.show(panelCentral, "menu"));
        }});
        sur.add(btnActualizar);
        sur.add(btnReg);

        Runnable refrescar = () -> {
            modelo.setRowCount(0);
            for (Producto p : productoServicio.getProductos())
                modelo.addRow(new Object[]{p.getCodigo(), p.getNombre(), p.getTipo(),
                        String.format("S/ %.2f", p.calcularPrecioFinal())});
        };

        btnReg.addActionListener(e -> {
            String cod    = txtCodigo.getText().trim();
            String nom    = txtNombre.getText().trim();
            String prec   = txtPrecio.getText().trim();
            String tipo   = (String) cbTipo.getSelectedItem();
            boolean extra = chkExtra.isSelected();
            String talla  = txtTalla.getText().trim();

            if (cod.isEmpty() || nom.isEmpty() || prec.isEmpty()) {
                JOptionPane.showMessageDialog(panel, "Codigo, Nombre y Precio son obligatorios.");
                return;
            }
            try {
                int    codigo = Integer.parseInt(cod);
                double precio = Double.parseDouble(prec);
                Producto p;
                if ("Electronico".equals(tipo))
                    p = new ProductoElectronico(codigo, nom, precio, extra);
                else if ("Alimento".equals(tipo))
                    p = new ProductoAlimento(codigo, nom, precio, extra);
                else {
                    if (talla.isEmpty()) { JOptionPane.showMessageDialog(panel, "Ingrese la talla."); return; }
                    p = new ProductoRopa(codigo, nom, precio, talla, extra);
                }
                productoServicio.registrar(p);
                JOptionPane.showMessageDialog(panel, "Producto registrado.");
                txtCodigo.setText(""); txtNombre.setText(""); txtPrecio.setText(""); txtTalla.setText("");
                chkExtra.setSelected(false);
                refrescar.run();
            } catch (NumberFormatException ex) {
                JOptionPane.showMessageDialog(panel, "Codigo y Precio deben ser numericos.");
            } catch (ProductoException | IllegalArgumentException ex) {
                JOptionPane.showMessageDialog(panel, ex.getMessage());
            }
        });

        btnActualizar.addActionListener(e -> refrescar.run());

        panel.add(form,   BorderLayout.NORTH);
        panel.add(scroll, BorderLayout.CENTER);
        panel.add(sur,    BorderLayout.SOUTH);

        refrescar.run();
        return panel;
    }

    // ============================================================
    //  P4 – REGISTRAR VENTA
    // ============================================================
    private JPanel panelVentas() {
        JPanel panel = new JPanel(new BorderLayout(5, 5));
        panel.setBorder(BorderFactory.createEmptyBorder(10, 10, 10, 10));

        // Cabecera
        JPanel cabecera = new JPanel(new FlowLayout(FlowLayout.LEFT, 8, 4));
        cabecera.setBorder(BorderFactory.createTitledBorder("Cabecera de la Venta"));
        JTextField txtCliente = new JTextField(8);
        JTextField txtFecha   = new JTextField(10);
        txtFecha.setText("2026-06-17");
        JLabel lblCliente = new JLabel("–");
        JButton btnBuscarCli = new JButton("Buscar cliente");
        cabecera.add(new JLabel("Cod. cliente:")); cabecera.add(txtCliente);
        cabecera.add(btnBuscarCli);                cabecera.add(lblCliente);
        cabecera.add(new JLabel("  Fecha:"));      cabecera.add(txtFecha);

        // Detalle
        JPanel detalle = new JPanel(new FlowLayout(FlowLayout.LEFT, 8, 4));
        detalle.setBorder(BorderFactory.createTitledBorder("Agregar Detalle"));
        JTextField txtProd = new JTextField(8);
        JTextField txtCant = new JTextField(5);
        JLabel lblProd = new JLabel("–");
        JButton btnBuscarProd = new JButton("Buscar producto");
        JButton btnAgregar    = new JButton("Agregar");
        detalle.add(new JLabel("Cod. producto:")); detalle.add(txtProd);
        detalle.add(btnBuscarProd);                detalle.add(lblProd);
        detalle.add(new JLabel("  Cantidad:"));    detalle.add(txtCant);
        detalle.add(btnAgregar);

        // Tabla de detalles
        String[] cols = {"Producto","Tipo","P. Unit.","Cant.","Subtotal"};
        DefaultTableModel modeloDet = new DefaultTableModel(cols, 0) {
            @Override public boolean isCellEditable(int r, int c) { return false; }
        };
        JTable tablaDet = new JTable(modeloDet);
        JScrollPane scrollDet = new JScrollPane(tablaDet);
        scrollDet.setBorder(BorderFactory.createTitledBorder("Detalle de la venta"));

        // Pie
        JLabel lblTotal = new JLabel("Total: S/ 0.00");
        JButton btnConfirmar = new JButton("Confirmar Venta");
        JButton btnCancelar  = new JButton("Cancelar");
        JButton btnMenu      = new JButton("Volver al menu");
        JPanel pie = new JPanel(new FlowLayout(FlowLayout.RIGHT));
        pie.add(lblTotal); pie.add(btnCancelar); pie.add(btnConfirmar); pie.add(btnMenu);

        // Estado
        final Venta[]   ventaActual = {null};
        final double[]  total       = {0};
        final Producto[] prodSel    = {null};

        Runnable resetVenta = () -> {
            ventaActual[0] = null; total[0] = 0;
            modeloDet.setRowCount(0); lblTotal.setText("Total: S/ 0.00");
            txtCliente.setText(""); lblCliente.setText("–");
            prodSel[0] = null; txtProd.setText(""); lblProd.setText("–"); txtCant.setText("");
        };

        btnBuscarCli.addActionListener(e -> {
            try {
                var c = clienteServicio.buscarPorCodigo(Integer.parseInt(txtCliente.getText().trim()));
                lblCliente.setText(c == null ? "No encontrado" : c.getNombre());
            } catch (NumberFormatException ex) { lblCliente.setText("Codigo invalido"); }
        });

        btnBuscarProd.addActionListener(e -> {
            try {
                Producto p = productoServicio.buscarPorCodigo(Integer.parseInt(txtProd.getText().trim()));
                prodSel[0] = p;
                lblProd.setText(p == null ? "No encontrado"
                        : p.getNombre() + " (S/ " + String.format("%.2f", p.calcularPrecioFinal()) + ")");
            } catch (NumberFormatException ex) { lblProd.setText("Codigo invalido"); }
        });

        btnAgregar.addActionListener(e -> {
            // Crear venta al agregar el primer detalle
            if (ventaActual[0] == null) {
                try {
                    int cod = Integer.parseInt(txtCliente.getText().trim());
                    var c   = clienteServicio.buscarPorCodigo(cod);
                    if (c == null) { JOptionPane.showMessageDialog(panel, "Cliente no encontrado."); return; }
                    contadorVentas++;
                    ventaActual[0] = new Venta(contadorVentas, c, txtFecha.getText().trim());
                } catch (NumberFormatException ex) {
                    JOptionPane.showMessageDialog(panel, "Busque primero un cliente valido."); return;
                }
            }
            if (prodSel[0] == null) { JOptionPane.showMessageDialog(panel, "Busque primero un producto."); return; }
            try {
                int cant = Integer.parseInt(txtCant.getText().trim());
                ventaServicio.agregarDetalle(ventaActual[0], prodSel[0], cant);
                double sub = prodSel[0].calcularPrecioFinal() * cant;
                total[0] += sub;
                modeloDet.addRow(new Object[]{
                    prodSel[0].getNombre(), prodSel[0].getTipo(),
                    String.format("S/ %.2f", prodSel[0].calcularPrecioFinal()),
                    cant, String.format("S/ %.2f", sub)
                });
                lblTotal.setText("Total: S/ " + String.format("%.2f", total[0]));
                prodSel[0] = null; txtProd.setText(""); lblProd.setText("–"); txtCant.setText("");
            } catch (NumberFormatException ex) {
                JOptionPane.showMessageDialog(panel, "La cantidad debe ser un numero entero.");
            } catch (VentaException ex) {
                JOptionPane.showMessageDialog(panel, ex.getMessage());
            }
        });

        btnConfirmar.addActionListener(e -> {
            if (ventaActual[0] == null || ventaActual[0].getDetalles().isEmpty()) {
                JOptionPane.showMessageDialog(panel, "Agregue al menos un producto."); return;
            }
            try {
                ventaServicio.registrarVenta(ventaActual[0]);
                JOptionPane.showMessageDialog(panel,
                        "Venta registrada. Total: S/ " + String.format("%.2f", ventaActual[0].getTotal()));
                resetVenta.run();
            } catch (VentaException ex) { JOptionPane.showMessageDialog(panel, ex.getMessage()); }
        });

        btnCancelar.addActionListener(e -> { resetVenta.run(); });
        btnMenu    .addActionListener(e -> cardLayout.show(panelCentral, "menu"));

        JPanel norte = new JPanel(new GridLayout(2, 1));
        norte.add(cabecera); norte.add(detalle);

        panel.add(norte,     BorderLayout.NORTH);
        panel.add(scrollDet, BorderLayout.CENTER);
        panel.add(pie,       BorderLayout.SOUTH);
        return panel;
    }

    // ============================================================
    //  P5 – LISTADO DE VENTAS
    // ============================================================
    private JPanel panelListaVentas() {
        JPanel panel = new JPanel(new BorderLayout(5, 5));
        panel.setBorder(BorderFactory.createEmptyBorder(10, 10, 10, 10));

        String[] cols = {"Cod.","Cliente","Fecha","Items","Total"};
        DefaultTableModel modelo = new DefaultTableModel(cols, 0) {
            @Override public boolean isCellEditable(int r, int c) { return false; }
        };
        JTable tabla = new JTable(modelo);
        JScrollPane scroll = new JScrollPane(tabla);
        scroll.setBorder(BorderFactory.createTitledBorder("Ventas registradas"));

        JTextArea areaDetalle = new JTextArea(6, 0);
        areaDetalle.setEditable(false);
        areaDetalle.setFont(new Font("Monospaced", Font.PLAIN, 11));
        JScrollPane scrollDet = new JScrollPane(areaDetalle);
        scrollDet.setBorder(BorderFactory.createTitledBorder("Detalle de la venta seleccionada"));

        JButton btnActualizar = new JButton("Actualizar lista");
        JButton btnMenu       = new JButton("Volver al menu");
        JPanel norte = new JPanel(new FlowLayout(FlowLayout.RIGHT));
        norte.add(btnMenu); norte.add(btnActualizar);

        Runnable refrescar = () -> {
            modelo.setRowCount(0);
            for (Venta v : ventaServicio.getVentas())
                modelo.addRow(new Object[]{
                    v.getCodigo(), v.getCliente().getNombre(), v.getFecha(),
                    v.getDetalles().size(), String.format("S/ %.2f", v.getTotal())
                });
            areaDetalle.setText("Haga clic en una venta para ver el detalle.");
        };

        tabla.getSelectionModel().addListSelectionListener(e -> {
            int row = tabla.getSelectedRow();
            if (row < 0) return;
            int cod = (int) modelo.getValueAt(row, 0);
            Venta v = ventaServicio.buscarPorCodigo(cod);
            if (v == null) return;
            StringBuilder sb = new StringBuilder();
            sb.append("Venta #").append(v.getCodigo())
              .append("  Cliente: ").append(v.getCliente().getNombre())
              .append("  Fecha: ").append(v.getFecha()).append("\n")
              .append("-".repeat(50)).append("\n");
            for (DetalleVenta d : v.getDetalles())
                sb.append(String.format("  %-20s x%d = S/ %.2f%n",
                        d.getProducto().getNombre(), d.getCantidad(), d.getSubtotal()));
            sb.append("-".repeat(50)).append("\n")
              .append(String.format("  TOTAL: S/ %.2f%n", v.getTotal()));
            areaDetalle.setText(sb.toString());
        });

        btnActualizar.addActionListener(e -> refrescar.run());
        btnMenu      .addActionListener(e -> cardLayout.show(panelCentral, "menu"));

        JSplitPane split = new JSplitPane(JSplitPane.VERTICAL_SPLIT, scroll, scrollDet);
        split.setResizeWeight(0.65);

        panel.add(norte, BorderLayout.NORTH);
        panel.add(split, BorderLayout.CENTER);

        refrescar.run();
        return panel;
    }

        // Datos de ejemplo
        try {
            clienteServicio.registrar(new Cliente(101, "Ana Torres", "ana@correo.com"));

            productoServicio.registrar(new ProductoAlimento(202, "Galletas", 2.50, true));

        } catch (Exception e) {
            System.out.println("Error cargando datos: " + e.getMessage());
        }

        SwingUtilities.invokeLater(() -> new SistemaRegistroUI().setVisible(true));
    }
